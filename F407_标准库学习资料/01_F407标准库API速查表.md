```c
#include "stm32f4xx_gpio.h"

void GPIO_Init(GPIO_TypeDef* GPIOx, GPIO_InitTypeDef* GPIO_InitStruct);
```

### USART初始化示例
```c
void USART3_Init(void)
{
    GPIO_InitTypeDef GPIO_InitStruct;
    USART_InitTypeDef USART_InitStruct;
    NVIC_InitTypeDef NVIC_InitStruct;

    // 1. 使能时钟
    RCC_AHB1PeriphClockCmd(RCC_AHB1Periph_GPIOD, ENABLE);
    RCC_APB1PeriphClockCmd(RCC_APB1Periph_USART3, ENABLE);

    // 2. 配置GPIO
    // PD8 -> USART3_TX
    GPIO_InitStruct.GPIO_Pin = GPIO_Pin_8;
    GPIO_InitStruct.GPIO_Mode = GPIO_Mode_AF;
    GPIO_InitStruct.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_InitStruct.GPIO_OType = GPIO_OType_PP;
    GPIO_InitStruct.GPIO_PuPd = GPIO_PuPd_NOPULL;
    GPIO_Init(GPIOD, &GPIO_InitStruct);
    GPIO_PinAFConfig(GPIOD, GPIO_PinSource8, GPIO_AF_USART3);

    // PD9 -> USART3_RX
    GPIO_InitStruct.GPIO_Pin = GPIO_Pin_9;
    GPIO_Init(GPIOD, &GPIO_InitStruct);
    GPIO_PinAFConfig(GPIOD, GPIO_PinSource9, GPIO_AF_USART3);

    // 3. 配置USART
    USART_InitStruct.USART_BaudRate = 115200;
    USART_InitStruct.USART_WordLength = USART_WordLength_8b;
    USART_InitStruct.USART_StopBits = USART_StopBits_1;
    USART_InitStruct.USART_Parity = USART_Parity_No;
    USART_InitStruct.USART_Mode = USART_Mode_Rx | USART_Mode_Tx;
    USART_InitStruct.USART_HardwareFlowControl = USART_HardwareFlowControl_None;
    USART_InitStruct.USART_Over = USART_Over_16;  // F4新增
    USART_Init(USART3, &USART_InitStruct);

    // 4. 使能USART
    USART_Cmd(USART3, ENABLE);

    // 5. 使能接收中断
    USART_ITConfig(USART3, USART_IT_RXNE, ENABLE);

    // 6. 配置NVIC
    NVIC_PriorityGroupConfig(NVIC_PriorityGroup_2);
    NVIC_InitStruct.NVIC_IRQChannel = USART3_IRQn;
    NVIC_InitStruct.NVIC_IRQChannelPreemptionPriority = 1;
    NVIC_InitStruct.NVIC_IRQChannelSubPriority = 0;
    NVIC_InitStruct.NVIC_IRQChannelCmd = ENABLE;
    NVIC_Init(&NVIC_InitStruct);
}
```

### 定时器计算
```
时钟频率 = 定时器时钟 / (Prescaler + 1)
定时时间 = (Period + 1) / 时钟频率
PWM频率 = 时钟频率 / (Period + 1)
占空比 = CompareValue / (Period + 1) × 100%

F407定时器时钟:
  APB1定时器 (TIM2-7, TIM12-14): 84MHz (APB1=42MHz, 自动倍频)
  APB2定时器 (TIM1, TIM8-11): 168MHz (APB2=84MHz, 自动倍频)

示例 (TIM2, 84MHz):
  Prescaler = 83  → 时钟频率 = 84MHz / 84 = 1MHz
  Period = 9999   → 定时 = 10000 / 1MHz = 10ms
  Compare = 5000  → PWM占空比 = 5000/10000 = 50%
```

### TIM初始化示例
```c
void TIM2_Init(void)
{
    TIM_TimeBaseInitTypeDef TIM_InitStruct;
    NVIC_InitTypeDef NVIC_InitStruct;

    RCC_APB1PeriphClockCmd(RCC_APB1Periph_TIM2, ENABLE);

    // 10ms定时: 84MHz / (83+1) / (9999+1) = 10ms
    TIM_InitStruct.TIM_Period = 9999;
    TIM_InitStruct.TIM_Prescaler = 83;
    TIM_InitStruct.TIM_ClockDivision = TIM_CKD_DIV1;
    TIM_InitStruct.TIM_CounterMode = TIM_CounterMode_Up;
    TIM_TimeBaseInit(TIM2, &TIM_InitStruct);

    TIM_ITConfig(TIM2, TIM_IT_Update, ENABLE);

    NVIC_PriorityGroupConfig(NVIC_PriorityGroup_2);
    NVIC_InitStruct.NVIC_IRQChannel = TIM2_IRQn;
    NVIC_InitStruct.NVIC_IRQChannelPreemptionPriority = 2;
    NVIC_InitStruct.NVIC_IRQChannelSubPriority = 0;
    NVIC_InitStruct.NVIC_IRQChannelCmd = ENABLE;
    NVIC_Init(&NVIC_InitStruct);

    TIM_Cmd(TIM2, ENABLE);
}
```

### ADC初始化示例
```c
void ADC1_Init(void)
{
    GPIO_InitTypeDef GPIO_InitStruct;
    ADC_InitTypeDef ADC_InitStruct;

    // 1. 使能时钟
    RCC_AHB1PeriphClockCmd(RCC_AHB1Periph_GPIOA, ENABLE);
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_ADC1, ENABLE);
    RCC_APB2PeriphClkConfig(RCC_APB2Periph_ADCCLK, RCC_PCLK2_Div4);
    // ADC时钟 = 84MHz/4 = 21MHz

    // 2. 配置GPIO为模拟输入
    GPIO_InitStruct.GPIO_Pin = GPIO_Pin_0;
    GPIO_InitStruct.GPIO_Mode = GPIO_Mode_AN;
    GPIO_Init(GPIOA, &GPIO_InitStruct);

    // 3. 配置ADC
    ADC_InitStruct.ADC_Mode = ADC_Mode_Independent;
    ADC_InitStruct.ADC_ScanConvMode = DISABLE;
    ADC_InitStruct.ADC_ContinuousConvMode = DISABLE;
    ADC_InitStruct.ADC_ExternalTrigConv = ADC_ExternalTrigConv_None;
    ADC_InitStruct.ADC_DataAlign = ADC_DataAlign_Right;
    ADC_InitStruct.ADC_ChannelCount = 1;
    ADC_Init(ADC1, &ADC_InitStruct);

    // 4. 配置通道
    ADC_ChannelConfig(ADC1, ADC_Channel_0, ADC_RegularChannelConfig_CHANNEL_0, 
                       ADC_SampleTime_84Cycles5);
    // F4中使用ADC_RegularChannelConfig:
    // ADC_RegularChannelConfig(ADC1, ADC_Channel_0, 1, ADC_SampleTime_84Cycles5);

    // 5. 使能ADC
    ADC_Cmd(ADC1, ENABLE);

    // 6. 校准
    ADC_ResetCalibration(ADC1);
    while (ADC_GetResetCalibrationStatus(ADC1) == SET);
    ADC_StartCalibration(ADC1);
    while (ADC_GetCalibrationStatus(ADC1) == SET);
}

uint16_t ADC1_Read_Value(void)
{
    ADC_SoftwareStartConv(ADC1);
    while (ADC_GetFlagStatus(ADC1, ADC_FLAG_EOC) == RESET);
    return ADC_GetConversionValue(ADC1);
}
```

## 九、Flash API

### F407 Flash结构
```
起始地址: 0x08000000
大小: 512KB (524288字节)

扇区结构 (F407VET6):
  Sector 0:  0x08000000 - 0x08003FFF (16KB)
  Sector 1:  0x08004000 - 0x08007FFF (16KB)
  Sector 2:  0x08008000 - 0x0800BFFF (16KB)
  Sector 3:  0x0800C000 - 0x0800FFFF (16KB)
  Sector 4:  0x08010000 - 0x0801FFFF (64KB)
  Sector 5:  0x08020000 - 0x0803FFFF (128KB)
  Sector 6:  0x08040000 - 0x0805FFFF (128KB)
  Sector 7:  0x08060000 - 0x0807FFFF (128KB)
```

### Flash写入示例
```c
#define FLASH_USER_ADDR  0x08060000  // Sector 6, 128KB

void Flash_WriteData(uint32_t addr, uint32_t *data, uint16_t len)
{
    uint16_t i;

    FLASH_Unlock();
    FLASH_ClearFlag(FLASH_FLAG_BSY | FLASH_FLAG_EOP);

    // 擦除扇区 (F4使用扇区擦除)
    FLASH_EraseSector(FLASH_Sector_6, FLASH_VOLTAGE_RANGE_3);
    while (FLASH_GetStatus() == FLASH_BUSY);

    // 写入数据 (F4支持双字写入)
    for (i = 0; i < len; i++) {
        FLASH_ProgramWord(addr + i * 4, data[i]);
        while (FLASH_GetStatus() == FLASH_BUSY);
    }

    FLASH_Lock();
}

void Flash_ReadData(uint32_t addr, uint32_t *data, uint16_t len)
{
    for (uint16_t i = 0; i < len; i++) {
        data[i] = *(__IO uint32_t*)(addr + i * 4));
    }
}
```

## 十一、常见问题

| 问题 | 解决方案 |
|------|----------|
| `stm32f4xx.h` 找不到 | 检查头文件路径，确保包含 `\Libraries\CMSIS\ST\STM32F4xx\Include` |
| 编译报 `undefined reference` | 检查是否在工程中添加了对应的 `.c` 文件 |
| GPIO 不工作 | 检查时钟是否使能、模式是否正确、AF配置是否正确 |
| USART 收不到数据 | 检查波特率、GPIO AF复用、USART_Cmd是否ENABLE |
| 中断不触发 | 检查NVIC优先级、EXTI配置、中断函数名是否正确 |
| TIM PWM 无输出 | 检查TIM_Cmd、OC配置、GPIO AF复用模式 |
| ADC 值不准 | 增加采样时间、校准ADC、检查阻抗 |
| Flash 写入失败 | 检查扇区大小、是否已解锁Flash、电压范围 |
| printf 无输出 | 检查重定向函数、USART配置 |
| F4特有外设编译错误 | 确保已包含对应头文件和源文件 |
