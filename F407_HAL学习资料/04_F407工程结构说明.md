## 二、关键文件说明

### 1. main.c - 主程序
**作用**: 系统入口，调用初始化函数和主循环

**关键结构**:
```c
/* USER CODE BEGIN PV */
// 在这里定义全局变量
volatile uint8_t user_flag = 0;
/* USER CODE END PV */

int main(void)
{
    HAL_Init();              // 初始化HAL库
    SystemClock_Config();    // 配置系统时钟
    
    MX_GPIO_Init();          // 初始化GPIO
    MX_USART3_UART_Init();   // 初始化USART3
    MX_TIM2_Init();          // 初始化TIM2
    MX_ADC1_Init();          // 初始化ADC1
    
    /* USER CODE BEGIN 2 */
    // 在这里添加你的初始化代码
    /* USER CODE END 2 */
    
    while (1) {
        /* USER CODE BEGIN WHILE */
        // 在这里添加你的主循环代码
        /* USER CODE END WHILE */
    }
}
```

**重要**: 业务代码必须写在 `USER CODE BEGIN` / `USER CODE END` 之间！

### 2. gpio.c - GPIO初始化
**作用**: 初始化GPIO引脚

**示例**:
```c
void MX_GPIO_Init(void)
{
    GPIO_InitTypeDef GPIO_InitStruct = {0};
    
    __HAL_RCC_GPIOC_CLK_ENABLE();
    __HAL_RCC_GPIOH_CLK_ENABLE();
    __HAL_RCC_GPIOA_CLK_ENABLE();
    __HAL_RCC_GPIOF_CLK_ENABLE();
    
    // 配置LED引脚 (PF9)
    HAL_GPIO_WritePin(GPIOF, GPIO_PIN_9, GPIO_PIN_RESET);
    
    GPIO_InitStruct.Pin = GPIO_PIN_9 | GPIO_PIN_10;
    GPIO_InitStruct.Mode = GPIO_MODE_OUTPUT_PP;
    GPIO_InitStruct.Pull = GPIO_NOPULL;
    GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_LOW;
    HAL_GPIO_Init(GPIOF, &GPIO_InitStruct);
    
    // 配置按键引脚 (PA0)
    GPIO_InitStruct.Pin = GPIO_PIN_0;
    GPIO_InitStruct.Mode = GPIO_MODE_INPUT;
    GPIO_InitStruct.Pull = GPIO_NOPULL;
    HAL_GPIO_Init(GPIOA, &GPIO_InitStruct);
    
    // 配置系统调试引脚 (PA13/PA14)
    // 通常由CubeMX自动处理
}
```

### 4. stm32f4xx_hal_msp.c - HAL MSP文件
**作用**: 底层硬件初始化 (时钟、中断、DMA等)

**MSP初始化函数示例**:
```c
void HAL_UART_MspInit(UART_HandleTypeDef* huart)
{
    GPIO_InitTypeDef GPIO_InitStruct = {0};
    
    if (huart->Instance == USART3)
    {
        // 1. 使能时钟
        __HAL_RCC_USART3_CLK_ENABLE();
        __HAL_RCC_GPIOD_CLK_ENABLE();
        
        // 2. 配置GPIO
        // PD8 -> USART3_TX
        GPIO_InitStruct.Pin = GPIO_PIN_8;
        GPIO_InitStruct.Mode = GPIO_MODE_AF_PP;
        GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_VERY_HIGH;
        HAL_GPIO_Init(GPIOD, &GPIO_InitStruct);
        
        // PD9 -> USART3_RX
        GPIO_InitStruct.Pin = GPIO_PIN_9;
        GPIO_InitStruct.Mode = GPIO_MODE_AF_PP;
        HAL_GPIO_Init(GPIOD, &GPIO_InitStruct);
        
        // 3. 配置NVIC
        HAL_NVIC_SetPriority(USART3_IRQn, 1, 0);
        HAL_NVIC_EnableIRQ(USART3_IRQn);
    }
}

void HAL_TIM_MspInit(TIM_HandleTypeDef* htim)
{
    if (htim->Instance == TIM2)
    {
        __HAL_RCC_TIM2_CLK_ENABLE();
        HAL_NVIC_SetPriority(TIM2_IRQn, 2, 0);
        HAL_NVIC_EnableIRQ(TIM2_IRQn);
    }
}

void HAL_ADC_MspInit(ADC_HandleTypeDef* hadc)
{
    GPIO_InitTypeDef GPIO_InitStruct = {0};
    
    if (hadc->Instance == ADC1)
    {
        __HAL_RCC_ADC1_CLK_ENABLE();
        __HAL_RCC_GPIOA_CLK_ENABLE();
        
        // PA0 -> ADC1_IN0
        GPIO_InitStruct.Pin = GPIO_PIN_0;
        GPIO_InitStruct.Mode = GPIO_MODE_ANALOG;
        HAL_GPIO_Init(GPIOA, &GPIO_InitStruct);
    }
}
```

## 三、代码分区规则

### CubeMX代码分区标记
```c
/* USER CODE BEGIN Includes */
#include "my_module.h"
/* USER CODE END Includes */

/* USER CODE BEGIN PV */
volatile uint8_t flag = 0;
/* USER CODE END PV */

/* USER CODE BEGIN 0 */
// 可被CubeMX重新生成的区域 (代码保留)
/* USER CODE END 0 */

/* USER CODE BEGIN 1 */
// 中断前代码
/* USER CODE END 1 */

/* USER CODE BEGIN 2 */
// 初始化后代码
/* USER CODE END 2 */

/* USER CODE BEGIN WHILE */
// 主循环代码
/* USER CODE END WHILE */

/* USER CODE BEGIN 3 */
// 主循环后代码
/* USER CODE END 3 */
```

### 注意事项
1. **业务代码必须写在 `USER CODE BEGIN` / `USER CODE END` 之间**
2. 重新运行CubeMX生成代码时，USER CODE区域内的代码会保留
3. USER CODE区域外的代码会被删除和覆盖
4. 建议将自定义模块放在单独的.c/.h文件中

### 推荐的代码组织方式
1. **Core**: CubeMX生成的初始化代码
2. **BSP**: 板级硬件抽象层 (LED、按键、串口等)
3. **Driver**: 通用外设驱动 (不依赖具体硬件)
4. **App**: 应用逻辑
5. **Middlewares**: 中间件 (协议、算法)
6. **Utils**: 通用工具

### 示例: BSP层组织
```c
// bsp_led.h
#ifndef BSP_LED_H
#define BSP_LED_H
#include "main.h"

void BSP_LED_Init(void);
void BSP_LED_ON(void);
void BSP_LED_OFF(void);
void BSP_LED_Toggle(void);

#endif

// bsp_led.c
#include "bsp_led.h"

void BSP_LED_Init(void)
{
    // 在USER CODE BEGIN 2中调用
    __HAL_RCC_GPIOF_CLK_ENABLE();
    
    GPIO_InitTypeDef GPIO_InitStruct = {0};
    GPIO_InitStruct.Pin = GPIO_PIN_9 | GPIO_PIN_10;
    GPIO_InitStruct.Mode = GPIO_MODE_OUTPUT_PP;
    GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_LOW;
    HAL_GPIO_Init(GPIOF, &GPIO_InitStruct);
    
    HAL_GPIO_WritePin(GPIOF, GPIO_PIN_9 | GPIO_PIN_10, GPIO_PIN_SET);
}

void BSP_LED_ON(void)
{
    HAL_GPIO_WritePin(GPIOF, GPIO_PIN_9, GPIO_PIN_RESET);
}

void BSP_LED_OFF(void)
{
    HAL_GPIO_WritePin(GPIOF, GPIO_PIN_9, GPIO_PIN_SET);
}

void BSP_LED_Toggle(void)
{
    HAL_GPIO_TogglePin(GPIOF, GPIO_PIN_9);
}
```
