## 项目一：电子秒表

### 功能要求
- OLED 显示时间（分:秒:毫秒）
- 按键控制：开始/停止/复位
- 精度：1ms

### 硬件连接
```
OLED (SSD1306 I2C)     STM32F103
- VCC  <----------> 3.3V
- GND  <----------> GND
- SCL  <----------> PB6 (I2C1_SCL)
- SDA  <----------> PB7 (I2C1_SDA)

按键 (KEY)
- 一端 <----------> PA0 (GPIOA)
- 另一端 <----------> GND
- 内部上拉（GPIO_PULLUP）
```

### 核心代码
```c
// stopwatch.h
typedef struct {
    uint32_t elapsed_ms;    // 累计毫秒
    uint32_t start_tick;    // 开始时的 tick
    uint8_t  running;       // 运行标志
    uint8_t  display_update;// 显示更新标志
} StopWatch_t;

// stopwatch.c
void StopWatch_Start(StopWatch_t *sw)
{
    if (!sw->running)
    {
        sw->start_tick = HAL_GetTick();
        sw->running = 1;
    }
}

void StopWatch_Stop(StopWatch_t *sw)
{
    if (sw->running)
    {
        sw->elapsed_ms += (HAL_GetTick() - sw->start_tick);
        sw->running = 0;
    }
}

void StopWatch_Reset(StopWatch_t *sw)
{
    sw->elapsed_ms = 0;
    sw->running = 0;
}

uint32_t StopWatch_GetTime(StopWatch_t *sw)
{
    if (sw->running)
    {
        return sw->elapsed_ms + (HAL_GetTick() - sw->start_tick);
    }
    return sw->elapsed_ms;
}

// main.c
StopWatch_t stopwatch = {0};
uint8_t last_key_state = 1;

void HAL_TIM_PeriodElapsedCallback(TIM_HandleTypeDef *htim)
{
    if (htim->Instance == TIM2)
    {
        static uint8_t tick_count = 0;
        tick_count++;
        if (tick_count >= 10)  // 10ms
        {
            tick_count = 0;
            stopwatch.display_update = 1;
        }
    }
}

int main(void)
{
    HAL_Init();
    SystemClock_Config();
    MX_GPIO_Init();
    MX_I2C1_Init();
    MX_TIM2_Init();  // 1ms 中断
    OLED_Init();
    
    HAL_TIM_Base_Start_IT(&htim2);
    
    OLED_Clear();
    OLED_ShowString(0, 0, "StopWatch");
    OLED_ShowString(0, 16, "00:00:000");
    
    while (1)
    {
        // 按键检测
        uint8_t key = HAL_GPIO_ReadPin(GPIOA, GPIO_PIN_0);
        if (key != last_key_state)
        {
            HAL_Delay(20);
            key = HAL_GPIO_ReadPin(GPIOA, GPIO_PIN_0);
            if (key != last_key_state)
            {
                if (key == GPIO_PIN_RESET)
                {
                    if (!stopwatch.running)
                        StopWatch_Start(&stopwatch);
                    else
                        StopWatch_Stop(&stopwatch);
                }
                last_key_state = key;
            }
        }
        
        // 更新显示
        if (stopwatch.display_update)
        {
            stopwatch.display_update = 0;
            uint32_t time = StopWatch_GetTime(&stopwatch);
            uint8_t min = time / 60000;
            uint8_t sec = (time / 1000) % 60;
            uint16_t ms = time % 1000;
            
            char buf[16];
            sprintf(buf, "%02d:%02d:%03d", min, sec, ms);
            OLED_ShowString(0, 32, buf);
        }
    }
}
```

## 项目二：智能温度监控

### 功能要求
- 读取 DHT11/DHT22 温湿度传感器
- OLED 实时显示
- 温度超过阈值蜂鸣器报警
- 按键调节报警阈值
- 数据存储到 Flash

### 硬件连接
```
DHT22          STM32F103
- VCC  <-----> 3.3V
- GND  <-----> GND
- DATA <-----> PB0 (GPIOA, 开漏)

蜂鸣器 (有源)
- 正极 <-----> PB1 (TIM3_CH4, PWM)
- 负极 <-----> GND
```

### 核心代码
```c
// main.c
#define DHT_PORT GPIOA
#define DHT_PIN  GPIO_PIN_0
#define BUZZER_PORT GPIOB
#define BUZZER_PIN  GPIO_PIN_1

typedef struct {
    float temperature;
    float humidity;
    float temp_alarm_threshold;
    float humidity_alarm_threshold;
    uint8_t alarm_enabled;
} Monitor_t;

Monitor_t monitor = {
    .temp_alarm_threshold = 35.0f,
    .humidity_alarm_threshold = 80.0f,
    .alarm_enabled = 1,
};

// DHT22 读取
uint8_t DHT22_Read(float *temp, float *hum)
{
    uint8_t data[5] = {0};
    uint8_t i, j;
    
    // 主机发送起始信号
    GPIO_InitTypeDef cfg = {0};
    cfg.Pin = DHT_PIN;
    cfg.Mode = GPIO_MODE_OUTPUT_PP;
    cfg.Speed = GPIO_SPEED_FREQ_HIGH;
    HAL_GPIO_Init(DHT_PORT, &cfg);
    
    HAL_GPIO_WritePin(DHT_PORT, DHT_PIN, GPIO_PIN_RESET);
    delay_us(20000);    // 拉低至少 18ms
    HAL_GPIO_WritePin(DHT_PORT, DHT_PIN, GPIO_PIN_SET);
    delay_us(30);       // 拉高 20~40us
    
    // 切换为输入
    cfg.Mode = GPIO_MODE_INPUT;
    cfg.Pull = GPIO_PULLUP;
    HAL_GPIO_Init(DHT_PORT, &cfg);
    
    // 等待 DHT 响应（低 80us + 高 80us）
    uint32_t timeout = 100;
    while (HAL_GPIO_ReadPin(DHT_PORT, DHT_PIN) && timeout--);
    timeout = 100;
    while (!HAL_GPIO_ReadPin(DHT_PORT, DHT_PIN) && timeout--);
    
    // 读取 40 位数据
    for (i = 0; i < 5; i++)
    {
        for (j = 0; j < 8; j++)
        {
            timeout = 100;
            while (!HAL_GPIO_ReadPin(DHT_PORT, DHT_PIN) && timeout--);
            delay_us(40);  // 50us 后判断
            data[i] <<= 1;
            if (HAL_GPIO_ReadPin(DHT_PORT, DHT_PIN))
                data[i] |= 1;
            timeout = 100;
            while (HAL_GPIO_ReadPin(DHT_PORT, DHT_PIN) && timeout--);
        }
    }
    
    // 校验
    if (data[4] == (uint8_t)(data[0] + data[1] + data[2] + data[3]))
    {
        *hum = (data[0] << 8 | data[1]) / 10.0f;
        *temp = (data[2] << 8 | data[3]) / 10.0f;
        return 0;
    }
    return 1;
}

// 蜂鸣器控制
void Buzzer_Beep(uint16_t duration_ms)
{
    HAL_GPIO_WritePin(BUZZER_PORT, BUZZER_PIN, GPIO_PIN_SET);
    HAL_Delay(duration_ms);
    HAL_GPIO_WritePin(BUZZER_PORT, BUZZER_PIN, GPIO_PIN_RESET);
}

// 主循环
while (1)
{
    if (DHT22_Read(&monitor.temperature, &monitor.humidity) == 0)
    {
        // 显示
        char buf[32];
        sprintf(buf, "T:%.1fC H:%.1f%%", monitor.temperature, monitor.humidity);
        OLED_ShowString(0, 0, buf);
        
        // 报警判断
        if (monitor.alarm_enabled && 
            (monitor.temperature > monitor.temp_alarm_threshold ||
             monitor.humidity > monitor.humidity_alarm_threshold))
        {
            Buzzer_Beep(200);
        }
    }
    
    HAL_Delay(2000);  // 每 2 秒读取一次
}
```

## 项目三：简易示波器

### 硬件连接
```
模拟输入      STM32F103
- 信号 <-----> PA0 (ADC1_CH0)
- GND  <-----> GND

OLED
- SCL  <-----> PB6
- SDA  <-----> PB7
```

### 核心代码
```c
#define SAMPLE_RATE 1000   // 1kHz 采样率
#define SAMPLE_SIZE 200    // 样本数

uint16_t adc_buffer[SAMPLE_SIZE];
uint8_t wave_buffer[64];   // OLED 显示缓冲区

// ADC 采样
void ADC_Sample(void)
{
    for (int i = 0; i < SAMPLE_SIZE; i++)
    {
        HAL_ADC_Start(&hadc1);
        HAL_ADC_PollForConversion(&hadc1, 10);
        adc_buffer[i] = HAL_ADC_GetValue(&hadc1);
        
        // 延时控制采样率
        HAL_Delay(1000 / SAMPLE_RATE);
    }
}

// 绘制波形到 OLED
void OLED_Draw_Wave(void)
{
    uint8_t x, y;
    uint16_t min_val = 4095, max_val = 0;
    uint8_t range;
    
    // 计算范围
    for (int i = 0; i < SAMPLE_SIZE; i++)
    {
        if (adc_buffer[i] < min_val) min_val = adc_buffer[i];
        if (adc_buffer[i] > max_val) max_val = adc_buffer[i];
    }
    range = (max_val - min_val) / 64 + 1;
    
    // 清屏
    OLED_Clear();
    
    // 绘制波形
    for (x = 0; x < 128 && x < SAMPLE_SIZE; x++)
    {
        y = 63 - (adc_buffer[x] - min_val) / range;
        OLED_DrawPoint(x, y);
    }
    
    // 显示信息
    char buf[32];
    float volt = (max_val - min_val) * 3.3f / 4096;
    sprintf(buf, "Vpp:%.2fV Freq:%dHz", volt, SAMPLE_RATE);
    OLED_ShowString(0, 0, buf);
    
    OLED_Display();
}

// 主循环
while (1)
{
    ADC_Sample();
    OLED_Draw_Wave();
}
```

## 项目四：智能密码锁

### 功能要求
- 4×4 矩阵键盘输入密码
- OLED 显示输入状态
- 密码正确解锁（LED 指示）
- 密码错误蜂鸣器报警
- 修改密码功能

### 硬件连接
```
4×4 矩阵键盘     STM32F103
- ROW0~3 <-----> PE0~PE3 (GPIOE, 输出)
- COL0~3 <-----> PE4~PE7 (GPIOE, 输入, 上拉)

LED (解锁)
- 正极 <-----> PD2 (GPIOD)

蜂鸣器
- 正极 <-----> PD3 (GPIOD)
```

### 核心代码
```c
#define ROW_PORT GPIOE
#define COL_PORT GPIOE
#define ROW_PINS (GPIO_PIN_0 | GPIO_PIN_1 | GPIO_PIN_2 | GPIO_PIN_3)
#define COL_PINS (GPIO_PIN_4 | GPIO_PIN_5 | GPIO_PIN_6 | GPIO_PIN_7)

#define PASSWORD_LENGTH 4
char current_password[PASSWORD_LENGTH] = "1234";
char new_password[PASSWORD_LENGTH] = "0000";
uint8_t input_buffer[PASSWORD_LENGTH];
uint8_t input_count = 0;

// 矩阵键盘扫描
uint8_t Keypad_Scan(void)
{
    uint8_t row, col;
    
    for (row = 0; row < 4; row++)
    {
        // 逐行扫描：拉低当前行
        GPIO_PinState row_pins[4] = {
            GPIO_PIN_RESET, GPIO_PIN_SET, GPIO_PIN_SET, GPIO_PIN_SET
        };
        HAL_GPIO_WritePin(ROW_PORT, row, row_pins[row]);
        HAL_Delay(1);  // 稳定
        
        // 检查列
        for (col = 0; col < 4; col++)
        {
            if (HAL_GPIO_ReadPin(COL_PORT, 4 + col) == GPIO_PIN_RESET)
            {
                // 返回键值
                // 行: 0~3, 列: 0~3
                return row * 4 + col + 1;
            }
        }
        
        // 恢复所有行为高
        for (int i = 0; i < 4; i++)
            HAL_GPIO_WritePin(ROW_PORT, i, GPIO_PIN_SET);
    }
    
    return 0;  // 无按键
}

// 主循环
while (1)
{
    uint8_t key = Keypad_Scan();
    
    if (key >= 1 && key <= 9)  // 数字键
    {
        if (input_count < PASSWORD_LENGTH)
        {
            input_buffer[input_count++] = key + '0';
            // 显示星号
            char display[PASSWORD_LENGTH + 1];
            memset(display, '*', input_count);
            display[input_count] = '\0';
            OLED_ShowString(0, 32, display);
        }
    }
    else if (key == 10)  // '*' 键 - 清除
    {
        input_count = 0;
        OLED_ShowString(0, 32, "****");
    }
    else if (key == 11)  // '#' 键 - 确认
    {
        if (input_count == PASSWORD_LENGTH)
        {
            if (memcmp(input_buffer, current_password, PASSWORD_LENGTH) == 0)
            {
                // 密码正确
                HAL_GPIO_WritePin(GPIOD, GPIO_PIN_2, GPIO_PIN_SET);  // LED 亮
                OLED_ShowString(0, 48, "UNLOCKED!");
                HAL_Delay(3000);
                HAL_GPIO_WritePin(GPIOD, GPIO_PIN_2, GPIO_PIN_RESET);
                OLED_ClearArea(0, 48, 128, 16);
            }
            else
            {
                // 密码错误
                HAL_GPIO_WritePin(GPIOD, GPIO_PIN_3, GPIO_PIN_SET);
                OLED_ShowString(0, 48, "WRONG!");
                HAL_Delay(500);
                HAL_GPIO_WritePin(GPIOD, GPIO_PIN_3, GPIO_PIN_RESET);
                OLED_ClearArea(0, 48, 128, 16);
            }
            input_count = 0;
        }
    }
}
```

## 项目五：蓝牙遥控小车

### 功能要求
- HC-05 蓝牙模块通信
- 手机 APP 发送控制命令
- 小车前进、后退、左转、右转、停止
- OLED 显示状态

### 硬件连接
```
HC-05           STM32F103
- VCC  <-----> 5V
- GND  <-----> GND
- TXD  <-----> PA3 (USART2_RX)
- RXD  <-----> PA2 (USART2_TX)

电机驱动 (L298N)
- IN1  <-----> PA4
- IN2  <-----> PA5
- IN3  <-----> PA6
- IN4  <-----> PA7
- ENA  <-----> PA8 (PWM)
- ENB  <-----> PA9 (PWM)
```

### 核心代码
```c
// 蓝牙命令解析
void Process_Bt_Command(char cmd)
{
    switch (cmd)
    {
        case 'F':  // 前进
            Motor_SetDirection(1, 1);  // IN1=1, IN2=0
            Motor_SetDirection(2, 1);  // IN3=1, IN4=0
            Motor_SetSpeed(200, 200);
            break;
        case 'B':  // 后退
            Motor_SetDirection(1, -1);
            Motor_SetDirection(2, -1);
            Motor_SetSpeed(200, 200);
            break;
        case 'L':  // 左转
            Motor_SetDirection(1, -1);
            Motor_SetDirection(2, 1);
            Motor_SetSpeed(150, 200);
            break;
        case 'R':  // 右转
            Motor_SetDirection(1, 1);
            Motor_SetDirection(2, -1);
            Motor_SetSpeed(200, 150);
            break;
        case 'S':  // 停止
            Motor_Stop();
            break;
        case 'U':  // 加速
            current_speed = 255;
            Motor_SetSpeed(current_speed, current_speed);
            break;
        case 'D':  // 减速
            current_speed = 100;
            Motor_SetSpeed(current_speed, current_speed);
            break;
    }
}

// 电机控制
void Motor_SetDirection(uint8_t motor, int8_t dir)
{
    if (motor == 1)
    {
        HAL_GPIO_WritePin(GPIOA, GPIO_PIN_4, dir > 0 ? GPIO_PIN_SET : GPIO_PIN_RESET);
        HAL_GPIO_WritePin(GPIOA, GPIO_PIN_5, dir > 0 ? GPIO_PIN_RESET : GPIO_PIN_SET);
    }
    else if (motor == 2)
    {
        HAL_GPIO_WritePin(GPIOA, GPIO_PIN_6, dir > 0 ? GPIO_PIN_SET : GPIO_PIN_RESET);
        HAL_GPIO_WritePin(GPIOA, GPIO_PIN_7, dir > 0 ? GPIO_PIN_RESET : GPIO_PIN_SET);
    }
}

void Motor_SetSpeed(uint16_t speed_l, uint16_t speed_r)
{
    __HAL_TIM_SET_COMPARE(&htim1, TIM_CHANNEL_1, speed_l);  // PA8
    __HAL_TIM_SET_COMPARE(&htim1, TIM_CHANNEL_2, speed_r);  // PA9
}

void Motor_Stop(void)
{
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_4, GPIO_PIN_RESET);
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_5, GPIO_PIN_RESET);
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_6, GPIO_PIN_RESET);
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_7, GPIO_PIN_RESET);
    Motor_SetSpeed(0, 0);
}

// USART2 接收处理
char bt_cmd = 0;
void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart)
{
    if (huart->Instance == USART2)
    {
        bt_cmd = received_byte;
        Process_Bt_Command(bt_cmd);
        
        // 显示状态
        char msg[20];
        sprintf(msg, "Cmd: %c Spd: %d", bt_cmd, current_speed);
        OLED_ShowString(0, 0, msg);
        
        // 继续接收
        HAL_UART_Receive_IT(&huart2, &received_byte, 1);
    }
}
```
