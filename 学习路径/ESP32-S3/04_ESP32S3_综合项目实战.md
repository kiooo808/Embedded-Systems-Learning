## 项目一：智能天气监测站

### 1.1 项目概述
```
功能：
1. 采集温湿度、光照、气压数据
2. Wi-Fi 连接路由器
3. OLED 实时显示数据
4. HTTP Server 提供网页查看
5. MQTT 上报数据到云端
6. 低功耗模式节省电量

硬件连接：
├── SHT30 (I2C) → PB6(SCL), PB7(SDA)
├── BH1750 (I2C) → 同上 I2C 总线
├── BMP280 (I2C) → 同上 I2C 总线
├── SSD1306 OLED (SPI) → PA7(MOSI), PA6(MISO), PA5(SCK), PE2(CS)
├── ESP32-S3 → Wi-Fi
└── USB → 调试
```

### 1.2 主程序结构
```c
// main/main.c
#include "driver/gpio.h"
#include "driver/i2c.h"
#include "driver/spi_master.h"
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "freertos/event_groups.h"
#include "esp_wifi.h"
#include "esp_http_server.h"
#include "mqtt_client.h"
#include "esp_log.h"

// 传感器驱动
#include "sht30.h"
#include "bh1750.h"
#include "bmp280.h"
#include "oled.h"

static const char *TAG = "WEATHER";

// 全局数据
typedef struct {
    float temperature;
    float humidity;
    float lux;
    float pressure;
} weather_data_t;

weather_data_t weather_data;
SemaphoreHandle_t data_mutex;
EventGroupHandle_t system_events;

#define WIFI_READY    BIT0
#define MQTT_READY    BIT1
#define OLED_UPDATE   BIT2
#define HTTP_UPDATE   BIT3

// 传感器任务
void sensor_task(void *param)
{
    while (1)
    {
        float temp, hum, lux, pres;
        
        // 读取所有传感器
        sht30_read(&temp, &hum);
        lux = bh1750_read_lux();
        bmp280_read_pressure(&pres);
        
        // 更新全局数据（加锁）
        xSemaphoreTake(data_mutex, portMAX_DELAY);
        weather_data.temperature = temp;
        weather_data.humidity = hum;
        weather_data.lux = lux;
        weather_data.pressure = pres;
        xSemaphoreGive(data_mutex);
        
        // 触发 OLED 和 HTTP 更新
        xEventGroupSetBits(system_events, OLED_UPDATE | HTTP_UPDATE);
        
        ESP_LOGI(TAG, "Temp: %.1fC, Hum: %.1f%%, Lux: %.0f, Pres: %.0fPa",
                 temp, hum, lux, pres);
        
        vTaskDelay(pdMS_TO_TICKS(2000));
    }
}

// OLED 显示任务
void oled_task(void *param)
{
    while (1)
    {
        xEventGroupWaitBits(system_events, OLED_UPDATE, 
                           pdTRUE, pdFALSE, portMAX_DELAY);
        
        // 读取数据（加锁）
        weather_data_t data;
        xSemaphoreTake(data_mutex, portMAX_DELAY);
        data = weather_data;
        xSemaphoreGive(data_mutex);
        
        // 更新 OLED
        oled_clear();
        oled_draw_string(0, 0, "Weather Station");
        oled_draw_string(0, 16, "Temp: %.1fC", data.temperature);
        oled_draw_string(0, 32, "Hum: %.1f%%", data.humidity);
        oled_draw_string(0, 48, "Lux: %.0f", data.lux);
        oled_display();
    }
}

// HTTP 服务器任务
httpd_handle_t http_server;

void start_http_server(void)
{
    httpd_config_t config = HTTPD_DEFAULT_CONFIG();
    config.server_port = 80;
    config.max_uri_handlers = 10;
    
    if (httpd_start(&http_server, &config) == ESP_OK)
    {
        // 注册处理器
        httpd_uri_t index_uri = {
            .uri = "/",
            .method = HTTP_GET,
            .handler = index_handler,
        };
        httpd_register_uri_handler(http_server, &index_uri);
        
        httpd_uri_t api_uri = {
            .uri = "/api/data",
            .method = HTTP_GET,
            .handler = api_data_handler,
        };
        httpd_register_uri_handler(http_server, &api_uri);
    }
}

static esp_err_t index_handler(httpd_req_t *req)
{
    // 返回 HTML 页面（包含 JavaScript 自动刷新）
    httpd_resp_set_type(req, "text/html");
    httpd_resp_send(req, html_page, HTTPD_RESP_USE_STRLEN);
    return ESP_OK;
}

static esp_err_t api_data_handler(httpd_req_t *req)
{
    weather_data_t data;
    xSemaphoreTake(data_mutex, portMAX_DELAY);
    data = weather_data;
    xSemaphoreGive(data_mutex);
    
    char json[100];
    snprintf(json, sizeof(json),
        "{\"temp\":%.1f,\"hum\":%.1f,\"lux\":%.0f,\"pres\":%.0f}",
        data.temperature, data.humidity, data.lux, data.pressure);
    
    httpd_resp_set_type(req, "application/json");
    httpd_resp_send(req, json, HTTPD_RESP_USE_STRLEN);
    return ESP_OK;
}

// MQTT 任务
void mqtt_task(void *param)
{
    while (1)
    {
        xEventGroupWaitBits(system_events, WIFI_READY, 
                           pdFALSE, pdFALSE, portMAX_DELAY);
        
        // MQTT 初始化与连接
        mqtt_init();
        
        while (1)
        {
            xEventGroupWaitBits(system_events, OLED_UPDATE,
                               pdTRUE, pdFALSE, portMAX_DELAY);
            
            // 读取数据并上报
            weather_data_t data;
            xSemaphoreTake(data_mutex, portMAX_DELAY);
            data = weather_data;
            xSemaphoreGive(data_mutex);
            
            char payload[150];
            snprintf(payload, sizeof(payload),
                "{\"temp\":%.1f,\"hum\":%.1f,\"lux\":%.0f,\"pres\":%.0f}",
                data.temperature, data.humidity, data.lux, data.pressure);
            
            esp_mqtt_client_publish(mqtt_client, "esp32s3/weather",
                                    payload, 0, 0, false);
        }
    }
}

void app_main(void)
{
    ESP_LOGI(TAG, "Weather Station Starting...");
    
    // 创建互斥锁和事件组
    data_mutex = xSemaphoreCreateMutex();
    system_events = xEventGroupCreate();
    
    // 初始化外设
    i2c_init();
    spi_init();
    oled_init();
    sht30_init();
    bh1750_init();
    bmp280_init();
    
    // 初始化 Wi-Fi
    wifi_init_sta();
    
    // 启动 HTTP Server
    start_http_server();
    xEventGroupSetBits(system_events, WIFI_READY);
    xEventGroupSetBits(system_events, HTTP_UPDATE);
    
    // 创建任务
    xTaskCreate(sensor_task, "sensor", 2048, NULL, 3, NULL);
    xTaskCreate(oled_task, "oled", 2048, NULL, 2, NULL);
    xTaskCreate(mqtt_task, "mqtt", 4096, NULL, 2, NULL);
    
    ESP_LOGI(TAG, "Weather Station Ready!");
}
```

### 2.2 主程序
```c
// main/main.c
#include "driver/adc.h"
#include "driver/spi_master.h"
#include "driver/gpio.h"
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "esp_log.h"

#define ADC_CHANNELS 4
#define ADC_SAMPLE_RATE 1000  // 1kHz

uint16_t adc_data[ADC_CHANNELS];
uint8_t usb_tx_buf[256];
volatile uint8_t sample_ready = 0;

// ADC + DMA 连续采样
void adc_dma_init(void)
{
    adc_continuous_handle_cfg_t handle_cfg = {
        .max_store_buf_size = 1024,
        .conv_mode = ADC_CONV_MODE_CONTINUOUS,
        .conv_frame_size = 100,
    };
    adc_continuous_new_handle(&handle_cfg, &adc_handle);
    
    adc_continuous_config_t cont_cfg = {
        .adc_mask = (1 << ADC_CHANNEL_10) | (1 << ADC_CHANNEL_11) |
                    (1 << ADC_CHANNEL_12) | (1 << ADC_CHANNEL_13),
        .atten = ADC_ATTEN_DB_12,
        .bitwidth = ADC_BITWIDTH_12,
        .sample_freq_hz = ADC_SAMPLE_RATE * 4,  // 每通道 1kHz
    };
    adc_continuous_config(adc_handle, &cont_cfg);
    adc_continuous_start(adc_handle);
}

// USB CDC 任务
void usb_task(void *param)
{
    while (1)
    {
        // 打包数据
        int len = 0;
        usb_tx_buf[len++] = 0xAA;  // 帧头
        usb_tx_buf[len++] = 0x55;
        usb_tx_buf[len++] = ADC_CHANNELS;
        usb_tx_buf[len++] = 0;  // 保留
        
        for (int i = 0; i < ADC_CHANNELS; i++)
        {
            usb_tx_buf[len++] = adc_data[i] & 0xFF;
            usb_tx_buf[len++] = (adc_data[i] >> 8) & 0xFF;
        }
        
        // 校验和
        uint8_t checksum = 0;
        for (int i = 0; i < len; i++) checksum += usb_tx_buf[i];
        usb_tx_buf[len++] = checksum;
        
        // 发送
        CDC_Transmit_FS(usb_tx_buf, len);
        
        vTaskDelay(pdMS_TO_TICKS(1));  // 1kHz
    }
}

// OLED 波形显示任务
void display_task(void *param)
{
    // 波形缓冲区
    uint8_t waveform[ADC_CHANNELS][128];  // 128 个采样点
    
    while (1)
    {
        // 滚动数据
        for (int ch = 0; ch < ADC_CHANNELS; ch++)
        {
            for (int i = 0; i < 127; i++)
                waveform[ch][i] = waveform[ch][i + 1];
            waveform[ch][127] = adc_data[ch] >> 4;  // 缩放到 0-15
        }
        
        // 绘制波形
        oled_clear();
        for (int ch = 0; ch < ADC_CHANNELS; ch++)
        {
            for (int i = 0; i < 127; i++)
            {
                int x = i;
                int y = 63 - waveform[ch][i];
                oled_draw_pixel(x, y);
                oled_draw_pixel(x + 1, 63 - waveform[ch][i + 1]);
            }
        }
        oled_display();
        
        vTaskDelay(pdMS_TO_TICKS(10));
    }
}

void app_main(void)
{
    // 初始化
    adc_dma_init();
    spi_init();
    oled_init();
    usb_init();
    
    // 创建任务
    xTaskCreate(usb_task, "usb", 4096, NULL, 3, NULL);
    xTaskCreate(display_task, "display", 4096, NULL, 2, NULL);
}
```

```c
// main/main.c

#define MATRIX_ROWS 4
#define MATRIX_COLS 4
#define PASSWORD_LENGTH 4
#define MAX_ATTEMPTS 3

// 密码存储（使用 NVS）
char current_password[PASSWORD_LENGTH + 1] = "1234";

// 状态机
typedef enum {
    STATE_IDLE,
    STATE_INPUT_PASSWORD,
    STATE_VERIFYING,
    STATE_UNLOCKED,
    STATE_ERROR,
    STATE_CHANGING_PASSWORD,
} lock_state_t;

lock_state_t current_state = STATE_IDLE;
char input_buffer[PASSWORD_LENGTH + 1];
int input_count = 0;
int error_count = 0;

// 矩阵键盘读取
char read_keypad(void)
{
    // 配置行为输出，列为输入
    // 扫描按键
    const char keymap[MATRIX_ROWS][MATRIX_COLS] = {
        {'1', '2', '3', 'A'},
        {'4', '5', '6', 'B'},
        {'7', '8', '9', 'C'},
        {'*', '0', '#', 'D'},
    };
    
    // 实现按键扫描...
    return 0;
}

// 验证密码
void verify_password(void)
{
    if (strcmp(input_buffer, current_password) == 0)
    {
        current_state = STATE_UNLOCKED;
        error_count = 0;
        unlock_door();
        show_message("UNLOCKED!");
    }
    else
    {
        error_count++;
        if (error_count >= MAX_ATTEMPTS)
        {
            current_state = STATE_ERROR;
            trigger_alarm();
            show_message("LOCKED!");
        }
        else
        {
            show_message("WRONG!");
        }
    }
    input_count = 0;
    memset(input_buffer, 0, sizeof(input_buffer));
}

// BLE 命令处理
void ble_command_handler(const char *cmd)
{
    if (strcmp(cmd, "unlock") == 0)
    {
        current_state = STATE_UNLOCKED;
        unlock_door();
        show_message("BLE UNLOCK!");
    }
    else if (strcmp(cmd, "status") == 0)
    {
        ble_send_status(current_state);
    }
}

// 主状态机
void state_machine_task(void *param)
{
    char key;
    uint32_t last_input = 0;
    
    while (1)
    {
        switch (current_state)
        {
            case STATE_IDLE:
                show_message("Enter Password");
                key = read_keypad();
                if (key)
                {
                    current_state = STATE_INPUT_PASSWORD;
                    input_count = 0;
                }
                break;
                
            case STATE_INPUT_PASSWORD:
                key = read_keypad();
                if (key >= '0' && key <= '9' && input_count < PASSWORD_LENGTH)
                {
                    input_buffer[input_count++] = key;
                    show_message(input_buffer);
                    last_input = esp_timer_get_time();
                }
                else if (key == 'A' || key == 'B')  // 确认
                {
                    verify_password();
                }
                else if (key == 'C')  // 取消
                {
                    current_state = STATE_IDLE;
                    input_count = 0;
                }
                
                // 超时
                if (esp_timer_get_time() - last_input > 10000000)
                {
                    current_state = STATE_IDLE;
                    input_count = 0;
                }
                break;
                
            case STATE_UNLOCKED:
                // 5 秒后自动上锁
                vTaskDelay(pdMS_TO_TICKS(5000));
                lock_door();
                current_state = STATE_IDLE;
                break;
                
            case STATE_ERROR:
                trigger_alarm();
                vTaskDelay(pdMS_TO_TICKS(30000));  // 锁定 30 秒
                current_state = STATE_IDLE;
                error_count = 0;
                break;
        }
        
        vTaskDelay(pdMS_TO_TICKS(50));
    }
}

void app_main(void)
{

    keypad_init();
    oled_init();
    relay_init();
    buzzer_init();
    ble_init();
    
    // 读取保存的密码

    // 启动状态机
    xTaskCreate(state_machine_task, "state", 4096, NULL, 3, NULL);
}
```

### 4.2 主程序
```c
// main/main.c

// 电机控制
#define MOTOR_A1 GPIO_NUM_4
#define MOTOR_A2 GPIO_NUM_5
#define MOTOR_B1 GPIO_NUM_6
#define MOTOR_B2 GPIO_NUM_7
#define MOTOR_PWM_A LEDC_CHANNEL_0
#define MOTOR_PWM_B LEDC_CHANNEL_1

typedef enum {
    CMD_FORWARD,
    CMD_BACKWARD,
    CMD_LEFT,
    CMD_RIGHT,
    CMD_STOP,
    CMD_SPEED_UP,
    CMD_SPEED_DOWN,
} car_cmd_t;

car_cmd_t current_cmd = CMD_STOP;
int current_speed = 50;  // 0-100

void motor_init(void)
{
    // GPIO 配置
    gpio_config_t io_conf = {
        .pin_bit_mask = (1 << MOTOR_A1) | (1 << MOTOR_A2) | 
                        (1 << MOTOR_B1) | (1 << MOTOR_B2),
        .mode = GPIO_MODE_OUTPUT,
    };
    gpio_config(&io_conf);
    
    // LEDC 配置
    ledc_timer_config_t timer_cfg = {
        .speed_mode = LEDC_LOW_SPEED_MODE,
        .duty_resolution = LEDC_TIMER_10_BIT,
        .timer_num = LEDC_TIMER_0,
        .freq_hz = 20000,
    };
    ledc_timer_config(&timer_cfg);
    
    ledc_channel_config_t chan_cfg = {
        .speed_mode = LEDC_LOW_SPEED_MODE,
        .timer_sel = LEDC_TIMER_0,
        .duty = 0,
    };
    
    chan_cfg.channel = MOTOR_PWM_A;
    chan_cfg.gpio_num = GPIO_NUM_15;
    ledc_channel_config(&chan_cfg);
    
    chan_cfg.channel = MOTOR_PWM_B;
    chan_cfg.gpio_num = GPIO_NUM_16;
    ledc_channel_config(&chan_cfg);
}

void car_move(car_cmd_t cmd)
{
    switch (cmd)
    {
        case CMD_FORWARD:
            gpio_set_level(MOTOR_A1, 1);
            gpio_set_level(MOTOR_A2, 0);
            gpio_set_level(MOTOR_B1, 1);
            gpio_set_level(MOTOR_B2, 0);
            break;
            
        case CMD_BACKWARD:
            gpio_set_level(MOTOR_A1, 0);
            gpio_set_level(MOTOR_A2, 1);
            gpio_set_level(MOTOR_B1, 0);
            gpio_set_level(MOTOR_B2, 1);
            break;
            
        case CMD_LEFT:
            gpio_set_level(MOTOR_A1, 1);
            gpio_set_level(MOTOR_A2, 0);
            gpio_set_level(MOTOR_B1, 0);
            gpio_set_level(MOTOR_B2, 1);
            break;
            
        case CMD_RIGHT:
            gpio_set_level(MOTOR_A1, 0);
            gpio_set_level(MOTOR_A2, 1);
            gpio_set_level(MOTOR_B1, 1);
            gpio_set_level(MOTOR_B2, 0);
            break;
            
        case CMD_STOP:
        default:
            gpio_set_level(MOTOR_A1, 0);
            gpio_set_level(MOTOR_A2, 0);
            gpio_set_level(MOTOR_B1, 0);
            gpio_set_level(MOTOR_B2, 0);
            break;
    }
    
    // 设置速度
    int duty = current_speed * 1023 / 100;
    ledc_set_duty(LEDC_LOW_SPEED_MODE, MOTOR_PWM_A, duty);
    ledc_set_duty(LEDC_LOW_SPEED_MODE, MOTOR_PWM_B, duty);
    ledc_update_duty(LEDC_LOW_SPEED_MODE, MOTOR_PWM_A);
    ledc_update_duty(LEDC_LOW_SPEED_MODE, MOTOR_PWM_B);
}

// 超声波避障
float read_distance(void)
{
    // 发送 10us 脉冲
    gpio_set_level(TRIG_PIN, 0);
    ets_delay_us(2);
    gpio_set_level(TRIG_PIN, 1);
    ets_delay_us(10);
    gpio_set_level(TRIG_PIN, 0);
    
    // 等待 Echo
    uint32_t start = esp_timer_get_time();
    while (gpio_get_level(ECHO_PIN) == 0)
    {
        if (esp_timer_get_time() - start > 100000) return -1;
    }
    
    start = esp_timer_get_time();
    while (gpio_get_level(ECHO_PIN) == 1)
    {
        if (esp_timer_get_time() - start > 100000) return -1;
    }
    
    uint32_t echo_time = esp_timer_get_time() - start;
    return echo_time * 0.0343f / 2.0f;  // cm
}

// 避障任务
void avoid_obstacle_task(void *param)
{
    while (1)
    {
        float distance = read_distance();
        
        if (distance > 0 && distance < 20.0f)  // 20cm 内有障碍
        {
            car_move(CMD_STOP);
            vTaskDelay(pdMS_TO_TICKS(500));
            
            // 后退一点
            car_move(CMD_BACKWARD);
            vTaskDelay(pdMS_TO_TICKS(500));
            car_move(CMD_STOP);
        }
        else if (current_cmd != CMD_STOP)
        {
            car_move(current_cmd);
        }
        
        vTaskDelay(pdMS_TO_TICKS(100));
    }
}

static esp_err_t web_index_handler(httpd_req_t *req)
{
    httpd_resp_set_type(req, "text/html");
    httpd_resp_send(req, web_html, HTTPD_RESP_USE_STRLEN);
    return ESP_OK;
}

static esp_err_t web_cmd_handler(httpd_req_t *req)
{
    char param[32];
    httpd_req_get_url_query_str(req, "f", param, sizeof(param));
    
    if (strcmp(param, "forward") == 0) current_cmd = CMD_FORWARD;
    else if (strcmp(param, "backward") == 0) current_cmd = CMD_BACKWARD;
    else if (strcmp(param, "left") == 0) current_cmd = CMD_LEFT;
    else if (strcmp(param, "right") == 0) current_cmd = CMD_RIGHT;
    else if (strcmp(param, "stop") == 0) current_cmd = CMD_STOP;
    
    httpd_resp_set_type(req, "application/json");
    httpd_resp_send(req, "{\"status\":\"ok\"}", HTTPD_RESP_USE_STRLEN);
    return ESP_OK;
}

    httpd_start(&server, &config);
    
    httpd_uri_t index_uri = {
        .uri = "/", .method = HTTP_GET, .handler = web_index_handler
    };
    httpd_register_uri_handler(server, &index_uri);
    
    httpd_uri_t cmd_uri = {
        .uri = "/cmd", .method = HTTP_GET, .handler = web_cmd_handler
    };
    httpd_register_uri_handler(server, &cmd_uri);
    
    // 创建任务
    xTaskCreate(avoid_obstacle_task, "avoid", 2048, NULL, 2, NULL);
    xTaskCreate(infrared_task, "ir", 1024, NULL, 2, NULL);
```
