## 第一节：项目一 - 以太网气象站

### 1.1 项目概述
```
功能：
1. 采集温湿度传感器（SHT30，I2C）
2. 采集光照传感器（BH1750，I2C）
3. 网页显示实时数据（LWIP + HTTP Server）
4. 支持远程查看（手机/电脑浏览器）

硬件连接：
├── SHT30 (I2C1)
│   ├── SCL: PB6
│   ├── SDA: PB7
│   └── ADDR: 0x44
├── BH1750 (I2C1)
│   ├── SCL: PB6
│   ├── SDA: PB7
│   └── ADDR: 0x23
├── Ethernet (RMII)
│   ├── PA1: MII_RX_CLK
│   ├── PA2: MII_TXD1/MDIO
│   ├── PA7: MII_RX_DV
│   ├── PC1: MDC
│   ├── PC2: RX_D0
│   ├── PC3: RX_D1
│   ├── PD3: TXD0
│   └── PD5: TX_EN
└── SPI1 (OLED 可选)
    ├── PA5: SCK
    ├── PA6: MISO
    ├── PA7: MOSI
    └── PE0: CS
```

### 2.1 功能概述

### 2.2 实现代码
```c
// main.c
#define ADC_CHANNELS 4
#define ADC_SAMPLE_RATE 1000  // 1kHz
#define USB_PACKET_SIZE 64

uint16_t adc_data[ADC_CHANNELS];
uint8_t usb_tx_buf[USB_PACKET_SIZE + 16];
volatile uint8_t adc_ready = 0;

// ADC + DMA 初始化
void ADC_Init(void)
{
    ADC_ChannelConfTypeDef sConfig = {0};
    
    hadc1.Instance = ADC1;
    hadc1.Init.ScanConvMode = ADC_SCAN_ENABLE;
    hadc1.Init.ContinuousConvMode = ENABLE;
    hadc1.Init.ExternalTrigConv = ADC_SOFTWARE_START;
    hadc1.Init.NbrOfConversion = ADC_CHANNELS;
    
    HAL_ADC_Init(&hadc1);
    
    // 配置 4 个通道
    for (int i = 0; i < ADC_CHANNELS; i++)
    {
        sConfig.Channel = ADC_CHANNEL_10 + i;  // PC0-PC3
        sConfig.Rank = ADC_REGULAR_RANK_1 + i;
        sConfig.SamplingTime = ADC_SAMPLETIME_84CYCLES_5;
        HAL_ADC_ConfigChannel(&hadc1, &sConfig);
    }
    
    // 启动 DMA
    HAL_ADC_Start_DMA(&hadc1, (uint32_t*)adc_data, ADC_CHANNELS);
}

// USB 数据包打包
void USB_Send_Data(void)
{
    int len = 0;
    
    // 协议头
    usb_tx_buf[len++] = 0xAA;  // Header
    usb_tx_buf[len++] = 0x55;
    usb_tx_buf[len++] = ADC_CHANNELS;  // Channel count
    usb_tx_buf[len++] = 0;  // Reserved
    
    // 数据（12-bit, 占 2 字节）
    for (int i = 0; i < ADC_CHANNELS; i++)
    {
        usb_tx_buf[len++] = adc_data[i] & 0xFF;
        usb_tx_buf[len++] = (adc_data[i] >> 8) & 0xFF;
    }
    
    // 校验和
    uint8_t checksum = 0;
    for (int i = 0; i < len; i++)
        checksum += usb_tx_buf[i];
    usb_tx_buf[len++] = checksum;
    
    // 发送
    CDC_Transmit_FS(usb_tx_buf, len);
}

// 主循环
int main(void)
{
    HAL_Init();
    SystemClock_Config();
    
    MX_GPIO_Init();
    MX_USB_OTG_FS_Init();
    ADC_Init();
    
    USB_CDC_Init();
    
    while (1)
    {
        // 等待 ADC 完成
        if (adc_ready)
        {
            USB_Send_Data();
            adc_ready = 0;
        }
        
        // 1kHz 采样间隔
        HAL_Delay(1);
    }
}
```

## 第三节：F407 低功耗设计

### 3.1 F407 低功耗模式
```
F407 低功耗模式：
├── Sleep Mode (睡眠)
│   - CPU 停止，外设运行
│   - 恢复时间: ~1 个时钟周期
│   - 功耗: ~100mA @ 168MHz (运行)
│   - 触发: WFI/WFE 指令
│
├── Stop Mode (停止)
│   - CPU + 大部分外设停止
│   - SRAM 保持
│   - 恢复时间: ~10us
│   - 功耗: ~15uA
│   - 触发: EXTI 中断
│
├── Standby Mode (待机)
│   - 所有时钟停止
│   - 仅备份寄存器 + SRAM 前 20 字节保持
│   - 恢复时间: ~100ms (冷启动)
│   - 功耗: ~0.5uA
│   - 触发: RTC 闹钟或外部复位
│
└── Under Reset (复位)
    - 所有状态丢失
    - 功耗: ~0.1uA
```

### 3.2 Stop Mode 实现
```c
// 进入 Stop Mode
void Enter_Stop_Mode(void)
{
    // 1. 关闭不需要的外设时钟
    __HAL_RCC_GPIOA_CLK_DISABLE();
    // ... 根据需要关闭
    
    // 2. 配置唤醒源（如 EXTI）
    GPIO_InitTypeDef GPIO_InitStruct = {0};
    GPIO_InitStruct.Pin = GPIO_PIN_0;  // PA0 作为唤醒
    GPIO_InitStruct.Mode = GPIO_MODE_IT_RISING;
    GPIO_InitStruct.Pull = GPIO_PULLDOWN;
    HAL_GPIO_Init(GPIOA, &GPIO_InitStruct);
    
    HAL_NVIC_SetPriority(EXTI0_IRQn, 2, 0);
    HAL_NVIC_EnableIRQ(EXTI0_IRQn);
    
    // 3. 进入 Stop 模式
    HAL_PWREx_EnterSTOPMode(PWR_STOPENTRY_WFI);
    
    // 4. 唤醒后恢复系统时钟
    SystemClock_Config();
    
    // 5. 重新初始化外设
    MX_GPIO_Init();
    MX_USART3_UART_Init();
    // ...
}

// EXTI 唤醒回调
void HAL_GPIO_EXTI_Callback(uint16_t GPIO_Pin)
{
    if (GPIO_Pin == GPIO_PIN_0)
    {
        // 唤醒处理
    }
}
```

### 3.3 Standby Mode + RTC 唤醒
```c
// RTC 配置（在 Standby 模式下定时唤醒）
void RTC_Init_Standby(void)
{
    RCC_OscInitTypeDef RCC_OscInitStruct = {0};
    RTC_TimeTypeDef RTC_TimeStruct = {0};
    RTC_DateTypeDef RTC_DateStruct = {0};
    RTC_AlarmTypeDef RTC_AlarmStruct = {0};
    
    // 配置 LSE
    RCC_OscInitStruct.OscillatorType = RCC_OSCILLATORTYPE_LSE;
    RCC_OscInitStruct.LSEState = RCC_LSE_ON;
    HAL_RCC_OscConfig(&RCC_OscInitStruct);
    
    // 使能 RTC 时钟
    __HAL_RCC_RTC_CLK_ENABLE();
    
    // 配置 RTC
    hrtc.Instance = RTC;
    hrtc.Init.HourFormat = RTC_HOURFORMAT_24;
    hrtc.Init.AsynchPrediv = 127;
    hrtc.Init.SynchPrediv = 255;
    HAL_RTC_Init(&hrtc);
    
    // 设置 RTC 闹钟
    RTC_AlarmStruct.AlarmTime.Hours = 0;
    RTC_AlarmStruct.AlarmTime.Minutes = 0;
    RTC_AlarmStruct.AlarmTime.Seconds = 0;
    RTC_AlarmStruct.AlarmMask = RTC_ALARMMASK_ALL;  // 每秒触发
    HAL_RTC_SetAlarm(&hrtc, &RTC_AlarmStruct, RTC_FORMAT_BIN);
    HAL_RTC_EnableAlarm(&hrtc);
    
    // 使能 RTC 闹钟中断
    HAL_NVIC_SetPriority(RTC_Alarm_IRQn, 2, 0);
    HAL_NVIC_EnableIRQ(RTC_Alarm_IRQn);
    
    // 使能 RTC Alarm 作为唤醒源
    __HAL_PWR_CLEAR_FLAG(PWR_FLAG_WU);
    HAL_PWR_EnableWakeUpPin(PWR_WAKEUP_PIN1);
}

// 进入 Standby 模式
void Enter_Standby_Mode(void)
{
    HAL_PWREx_EnterSTANDBYMode(PWR_STANDBYENTRY_WFI);
    // 唤醒后相当于复位
    // 需要重新执行初始化
}
```

### 4.3 HardFault 调试
```c
// 在 main.c 或独立文件中添加
// 用于捕获 HardFault

void HardFault_Handler(void)
{
    // 获取堆栈信息
    uint32_t sp, pc, lr, psr;
    
    // 从 MSP 获取
    if (__get_MSP() < 0x20000000 || __get_MSP() > 0x20020000)
    {
        sp = __get_PSP();
    }
    else
    {
        sp = __get_MSP();
    }
    
    // 打印调试信息（如果串口可用）
    printf("HardFault!\n");
    printf("SP: 0x%08lX\n", sp);
    printf("PC: 0x%08lX\n", __get_PC());
    printf("LR: 0x%08lX\n", __get_LR());
    printf("PSR: 0x%08lX\n", __get_xPSR());
    
    // 打印堆栈内容
    uint32_t *stack = (uint32_t*)sp;
    for (int i = 0; i < 10; i++)
    {
        printf("  [%d]: 0x%08lX\n", i, stack[i]);
    }
    
    while (1);
}

// 使用 MemManage Fault 捕获更多信息
void MemManage_Handler(void)
{

    printf("CFSR: 0x%08lX\n", SCB->CFSR);
    printf("MMFAR: 0x%08lX\n", SCB->MMFAR);
```

## 第五节：F407 常见问题

### 5.1 时钟相关问题
```
Q: F407 串口波特率不对？
A: 
1. 检查 HSE 频率（通常是 8MHz）
2. 检查 PLL 配置
3. 确认实际时钟：SystemCoreClockUpdate()
4. 检查 APB1/APB2 分频

Q: TIM PWM 频率不对？
A: 
1. F407 定时器时钟会自动倍频！
   - APB1=42MHz → TIM2-7 = 84MHz
   - APB2=84MHz → TIM1,8 = 168MHz
2. 分频系数要对应
3. 例如 TIM3 (84MHz): Prescaler=83, Period=999 → 1kHz

Q: ADC 转换速度慢？
A: 
1. 缩短采样时间
2. 使用 DMA + 连续转换
3. ADC 时钟 = APB2/4 = 21MHz
```

### 5.2 Flash 相关问题
```
Q: Flash 擦除函数报错？
A: 
F407 使用扇区擦除，不是页擦除！

// F103: FLASH_PageErase(Page)
// F407: FLASH_Erase_Sector(Sector, VoltageRange)

// F407 擦除示例
FLASH_EraseInitTypeDef erase = {0};
erase.TypeErase = FLASH_TYPEERASE_SECTORS;
erase.Sector = FLASH_SECTOR_5;  // 128KB 扇区
erase.NbSectors = 1;
erase.VoltageRange = FLASH_VOLTAGE_RANGE_3;  // 2.7-3.6V
HAL_FLASHEx_Erase(&erase, &page_error);

// F407 Flash 结构
// Sector 0-3: 16KB
// Sector 4: 64KB
// Sector 5+: 128KB

Q: Flash 写入数据不对？
A: 
1. F407 必须按 256 页编程（或按页编程）
2. 先解锁：HAL_FLASH_Unlock()
3. 擦除对应扇区
4. 编程：HAL_FLASH_Program()
5. 锁定：HAL_FLASH_Lock()

// 正确的 F407 Flash 编程
HAL_FLASH_Unlock();

FLASH_EraseInitTypeDef erase = {0};
erase.TypeErase = FLASH_TYPEERASE_SECTORS;
erase.Sector = FLASH_SECTOR_5;
erase.NbSectors = 1;
HAL_FLASHEx_Erase(&erase, &page_error);

for (int i = 0; i < len; i++)
{
    HAL_FLASH_Program(FLASH_TYPEPROGRAM_DOUBLEWORD, 
        address + i * 8, *(uint64_t*)(data + i * 8));
}

HAL_FLASH_Lock();
```

### 5.3 GPIO 相关问题
```
Q: F407 GPIO 输出不工作？
A: 
1. 检查 GPIO 时钟是否使能
2. F407 必须配置 Speed 字段！
3. 检查引脚是否复用给其他外设
4. 检查 VDD 供电

Q: USB CDC 不枚举？
A: 
1. PA9 必须配置为 VBUS 检测输入
2. USB 时钟必须为 48MHz (PLLQ=7)
3. 检查 USB D+/D- 连接
4. 重写 USB 描述符（VID/PID）
5. 在设备管理器中检查是否有未知设备
```

### 5.4 Ethernet 相关问题
```
Q: Ethernet 无法连接？
A: 
1. 检查 RMII 引脚连接
2. 检查 PHY 地址（通常是 0）
3. 配置 MAC 地址（不能全零）
4. 等待 PHY 就绪（检查 BSR 寄存器）
5. 检查网线和 HUB/switch

Q: LWIP 初始化失败？
A: 
1. 需要先初始化 ETH
2. 注册 netif 时检查返回值
3. 确保 tcpip 线程在运行（FreeRTOS）
4. 使用 ping 命令测试连接
```

### 5.5 SDIO 相关问题
```
Q: SD 卡无法挂载？
A: 
1. 检查 SDIO 引脚连接
2. SDIO 时钟频率不要超过 400kHz（初始化时）
3. 使用 CMD0 (GO_IDLE_STATE) 检查响应
4. 检查 SD 卡类型（V1/V2/SDHC）
5. 使用 SDIO_DMA 模式传输

Q: SD 卡读写慢？
A: 
1. 使用 4-bit 数据线模式
2. 使用 DMA 传输
3. 增大块大小（512→4096）
4. 使用 SDIO 高速模式（HS）
```

### 5.6 FPU 相关问题
```
Q: FPU 代码无法编译？
A: 
1. 在 CubeMX 中启用 FPU
2. 编译器选项：-mfpu=fpv4-sp -mfloat-abi=hard
3. 添加 FPU 初始化代码
4. 确保使用 float 而非 double

// CubeMX 自动生成的 FPU 配置
// SystemClock_Config() 后：
SCB->CPACR |= ((3UL << 10*2)|(3UL << 11*2));

// 编译器选项（Keil）
// Options → Target → FPU: Single Precision
// Options → Target → Floating Point: Single
```
