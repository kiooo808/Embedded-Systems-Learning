### 1.2 F407 GPIO 模式详解
```
F407 GPIO 模式（比 F103 更丰富）:
├── 输入模式 (Input)
│   ├── 浮空输入 (Floating)
│   ├── 上拉输入 (Pull-up)
│   └── 下拉输入 (Pull-down)
├── 输出模式 (Output)
│   ├── 推挽输出 (Push-Pull)
│   └── 开漏输出 (Open-Drain)
├── 复用模式 (Alternate Function)
│   ├── 复用推挽 (AF_PP)
│   └── 复用开漏 (AF_OD)
└── 模拟模式 (Analog)
```

void USART3_IRQHandler(void)
{
    // 空闲中断
    if (__HAL_UART_GET_FLAG(&huart3, UART_FLAG_IDLE))
    {
        __HAL_UART_CLEAR_IDLEFLAG(&huart3);
        
        // 停止 DMA
        HAL_UART_DMAStop(&huart3);
        
        // 获取已接收长度
        uint16_t rx_len = sizeof(rx_buf) - 
            __HAL_DMA_GET_COUNTER(huart3.hdmarx);
        
        // 处理数据
        Process_UART_Data(rx_buf, rx_len);
        
        // 重启 DMA 接收
        memset(rx_buf, 0, sizeof(rx_buf));
        HAL_UART_Receive_DMA(&huart3, rx_buf, sizeof(rx_buf));
    }
    
    HAL_UART_IRQHandler(&huart3);
}

### 2.3 UART 环形缓冲接收
```c
// 使用环形缓冲区实现高效接收
#define RING_BUF_SIZE 1024

typedef struct {
    uint8_t buffer[RING_BUF_SIZE];
    volatile uint16_t head;   // 写入位置
    volatile uint16_t tail;   // 读取位置
} RingBuffer_t;

RingBuffer_t rx_ring;

// DMA 接收（指定长度为 1，实现逐字节接收）
void UART_Ring_Init(void)
{
    rx_ring.head = 0;
    rx_ring.tail = 0;
    HAL_UART_Receive_DMA(&huart3, &rx_ring.buffer[0], 1);
}

// DMA 完成回调（每接收 1 字节触发）
void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart)
{
    if (huart->Instance == USART3)
    {
        // 移动 head
        rx_ring.head = (rx_ring.head + 1) % RING_BUF_SIZE;
        
        // 重启 DMA（接收下一字节到新位置）
        HAL_UART_Receive_DMA(&huart3, 
            &rx_ring.buffer[rx_ring.head], 1);
    }
}

// 读取环形缓冲
uint16_t Ring_Read(uint8_t *data, uint16_t max_len)
{
    uint16_t count = 0;
    while (rx_ring.tail != rx_ring.head && count < max_len)
    {
        data[count++] = rx_ring.buffer[rx_ring.tail];
        rx_ring.tail = (rx_ring.tail + 1) % RING_BUF_SIZE;
    }
    return count;
}
```

## 第三节：TIM 与 PWM（F407 增强）

### 3.1 F407 定时器资源
```
F103: 7 个定时器
F407: 14+ 个定时器
├── 基本: TIM6, TIM7（仅基本定时）
├── 通用: TIM2-TIM5, TIM12-TIM15
├── 高级: TIM1, TIM8（互补输出、死区）
└── 特殊: TIM9-TIM11（单通道）

定时器时钟:
├── APB1 = 42MHz → TIM2-7, TIM12-14 = 84MHz (自动×2)
├── APB2 = 84MHz → TIM1, TIM8, TIM9-11 = 168MHz (自动×2)
└── 当 APBx 分频时，定时器时钟自动倍频！
```

## 第四节：ADC（多通道 + DMA）

### 4.1 F407 ADC 架构
```
F407 有 3 个 ADC:
├── ADC1: 16 通道 (0~15)
├── ADC2: 16 通道 (0~15)  
└── ADC3: 8 通道 (0~7)

ADC 时钟: APB2 = 84MHz, ADC 时钟 = 84MHz/4 = 21MHz
转换时间: 
  - 12-bit: 12 cycles + sampling time
  - 最小: 12 + 4 = 16 cycles (0.76us)
  - 常用: 12 + 84 = 96 cycles (4.57us)
```

    HAL_ADCEx_EnableTemperatureSensor();

}

// 读取温度
float ADC_Get_Temperature(void)
{
    uint16_t raw;
    HAL_ADC_Start(&hadc1);
    HAL_ADC_PollForConversion(&hadc1, 100);
    raw = HAL_ADC_GetValue(&hadc1);
    
    // STM32F407 温度传感器公式
    // V_SENSE = 3.3V * raw / 4095
    // Temperature = (V_SENSE - V_25) / Avg_Slope + 25
    // V_25 ≈ 1.43V, Avg_Slope ≈ 0.0042V/°C
    float v_sense = 3.3f * raw / 4095.0f;
    float temp = (v_sense - 1.43f) / 0.0042f + 25.0f;
    return temp;
}

sdmmc_slot_config_t slot_config = SDMMC_SLOT_CONFIG_DEFAULT();
sdmmc_card_t *card;

void SDIO_Init(void)
{
    // 挂载 FATFS
    esp_vfs_fat_mount_config_t mount_config = {
        .format_if_mount_failed = true,
        .max_files = 5,
    };
    
    esp_err_t err = esp_vfs_fat_sdmmc_mount("/sdcard", &host, 
                                            &slot_config, &mount_config);
    if (err != ESP_OK)
    {
        ESP_LOGI("SDIO", "Mount failed: %s", esp_err_to_name(err));
        return;
    }
    
    ESP_LOGI("SDIO", "SD card mounted!");
    ESP_LOGI("SDIO", "Card size: %llu MB", (unsigned long long)card->csd.capacity);
}

// SD 卡读写
void SDIO_Example(void)
{
    // 写文件
    FILE *f = fopen("/sdcard/test.txt", "w");
    if (f) {
        fprintf(f, "Hello SD Card!\n");
        fclose(f);
    }
    
    // 读文件
    f = fopen("/sdcard/test.txt", "r");
    if (f) {
        char buf[256];
        fgets(buf, sizeof(buf), f);
        ESP_LOGI("SDIO", "Read: %s", buf);
        fclose(f);
    }
}
